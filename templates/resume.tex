% Custom commands
\newcommand{\resumeItem}[1]{\item\small{#1}\vspace{-2pt}}

\newcommand{\resumeEntry}[4]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-6pt}
}

\newcommand{\resumeProjectEntry}[2]{
  \vspace{-1pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \small\textbf{#1} & #2 \\
    \end{tabular*}\vspace{-6pt}
}

\newcommand{\resumeEntryStart}{\begin{itemize}[leftmargin=0.0in, label={}]}
\newcommand{\resumeEntryEnd}{\end{itemize}}
\newcommand{\resumeItemStart}{\begin{itemize}[leftmargin=0.15in]}
\newcommand{\resumeItemEnd}{\end{itemize}\vspace{-4pt}}
